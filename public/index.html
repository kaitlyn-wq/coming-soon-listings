const grid = document.getElementById('listingsGrid'); const emptyState = document.getElementById('emptyState'); const modal = document.getElementById('formModal'); const form = document.getElementById('listingForm'); const formError = document.getElementById('formError'); const modalTitle = document.getElementById('modalTitle'); const submitBtn = document.getElementById('submitBtn'); const photosLabel = document.getElementById('photosLabel'); const existingPhotosEl = document.getElementById('existingPhotos'); let listingsCache = []; let editingId = null; // null = adding a new listing, otherwise the id being edited let removedPhotos = new Set(); // filenames marked for removal while editing document.getElementById('openFormBtn').addEventListener('click', () => { openAddModal(); }); document.getElementById('closeFormBtn').addEventListener('click', closeModal); document.getElementById('cancelFormBtn').addEventListener('click', closeModal); modal.addEventListener('click', (e) => { if (e.target === modal) closeModal(); }); function openAddModal() { editingId = null; removedPhotos = new Set(); form.reset(); formError.hidden = true; modalTitle.textContent = 'Add a Coming Soon Listing'; submitBtn.textContent = 'Post Listing'; photosLabel.textContent = 'Photos (up to 8)'; existingPhotosEl.hidden = true; existingPhotosEl.innerHTML = ''; modal.hidden = false; } function openEditModal(listing) { editingId = listing.id; removedPhotos = new Set(); formError.hidden = true; form.address.value = listing.address || ''; form.city.value = listing.city || ''; form.state.value = listing.state || ''; form.price.value = listing.price || ''; form.listDate.value = listing.listDate || ''; form.highlights.value = (listing.highlights || []).join('\n'); form.agentName.value = listing.agentName || ''; form.photos.value = ''; modalTitle.textContent = 'Edit Listing'; submitBtn.textContent = 'Save Changes'; photosLabel.textContent = 'Add More Photos (up to 8 total)'; renderExistingPhotos(listing.photos || []); modal.hidden = false; } function renderExistingPhotos(photos) { if (!photos.length) { existingPhotosEl.hidden = true; existingPhotosEl.innerHTML = ''; return; } existingPhotosEl.hidden = false; existingPhotosEl.innerHTML = '

Current photos (click to remove)

' + '
' + photos .map( (filename) => `
Remove
` ) .join('') + '
'; existingPhotosEl.querySelectorAll('.existing-photo-thumb').forEach((thumb) => { thumb.addEventListener('click', () => { const filename = thumb.dataset.filename; if (removedPhotos.has(filename)) { removedPhotos.delete(filename); thumb.classList.remove('marked-removed'); } else { removedPhotos.add(filename); thumb.classList.add('marked-removed'); } }); }); } function closeModal() { modal.hidden = true; form.reset(); formError.hidden = true; existingPhotosEl.hidden = true; existingPhotosEl.innerHTML = ''; editingId = null; removedPhotos = new Set(); } function formatDate(isoDate) { if (!isoDate) return ''; const [y, m, d] = isoDate.split('-').map(Number); const dt = new Date(y, m - 1, d); return dt.toLocaleDateString(undefined, { month: 'short', day: 'numeric', year: 'numeric' }); } function renderListings(listings) { listingsCache = listings; grid.innerHTML = ''; emptyState.hidden = listings.length !== 0; for (const listing of listings) { const card = document.createElement('div'); card.className = 'listing-card'; let photosHtml = '
No photos yet
'; if (listing.photos && listing.photos.length) { const shown = listing.photos.slice(0, 3); const single = shown.length === 1; photosHtml = `
` + shown.map((f, i) => ``).join('') + `
`; } const highlightsHtml = (listing.highlights || []).length ? `
    ${listing.highlights.map((h) => `
  • ${escapeHtml(h)}
    • `).join('')}
` : ''; const locationParts = [listing.city, listing.state].filter(Boolean).join(', '); card.innerHTML = ` ${photosHtml}

${escapeHtml(listing.address)}

${locationParts ? `

${escapeHtml(locationParts)}

` : ''} ${listing.price ? `

${escapeHtml(listing.price)}

` : ''}

Coming soon — lists ${formatDate(listing.listDate)}

${highlightsHtml} ${listing.agentName ? `

Posted by ${escapeHtml(listing.agentName)}

` : ''}
`; grid.appendChild(card); } } grid.addEventListener('click', async (e) => { const btn = e.target.closest('button[data-action]'); if (!btn) return; const id = btn.dataset.id; const listing = listingsCache.find((l) => l.id === id); if (!listing) return; if (btn.dataset.action === 'edit') { openEditModal(listing); } else if (btn.dataset.action === 'delete') { const ok = confirm(`Remove the listing at ${listing.address}? This can't be undone.`); if (!ok) return; try { const res = await fetch(`/api/listings/${id}`, { method: 'DELETE' }); if (!res.ok) throw new Error('Failed to delete listing.'); await loadListings(); } catch (err) { alert(err.message); } } }); function escapeHtml(str) { const div = document.createElement('div'); div.textContent = str; return div.innerHTML; } async function loadListings() { try { const res = await fetch('/api/listings'); const listings = await res.json(); renderListings(listings); } catch (err) { console.error('Failed to load listings', err); } } form.addEventListener('submit', async (e) => { e.preventDefault(); formError.hidden = true; const formData = new FormData(form); if (editingId) { formData.append('removePhotos', Array.from(removedPhotos).join(',')); } const url = editingId ? `/api/listings/${editingId}` : '/api/listings'; const method = editingId ? 'PUT' : 'POST'; try { const res = await fetch(url, { method, body: formData }); if (!res.ok) { const data = await res.json().catch(() => ({})); throw new Error(data.error || 'Failed to save listing.'); } closeModal(); await loadListings(); } catch (err) { formError.textContent = err.message; formError.hidden = false; } }); loadListings(); // Refresh periodically so everyone's view stays current without a manual reload setInterval(loadListings, 60 * 1000);
